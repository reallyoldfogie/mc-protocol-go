## Overview
Currently, integration and bitfield test files are manually created for version 1.21.5. These tests should be generated for each Minecraft version to ensure all generated code is properly tested.

## Current State

### Test Files (Currently in `data/1.21.5/`)
1. **`integration_test.go`** - Tests generated types:
   - Simple containers (Vec3f)
   - Bitfields (Position)
   - VarInt fields (StepTick)
   - Complex containers (EntityTeleport)
   - Mapper types (EntityUpdateAttributesArrayTypeKey)
   - Switch fields (ScoreboardObjective)
   - Option types
   - Byte count accuracy

2. **`bitfield_test.go`** - Tests bitfield implementation:
   - Position encoding/decoding
   - Round-trip serialization
   - Positive, negative, and zero values

### Issues
- Tests are hardcoded for version 1.21.5
- Tests reference specific packet types that may not exist in other versions
- New versions require manually creating/updating test files
- Different versions may have different available types to test

```go
const bitfieldTestTemplate = `// Code generated by generator. DO NOT EDIT.
package {{.PackageName}}_test

import (
	"bytes"
	"testing"

	"github.com/reallyoldfogie/mc-protocol-go/data/{{.Version}}/basetypes"
)

func TestPositionBitfield(t *testing.T) {
	tests := []struct {
		name string
		pos  basetypes.Position
		want []byte
	}{
		{
			name: "zero values",
			pos:  basetypes.Position{X: 0, Z: 0, Y: 0},
			want: []byte{0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00},
		},
		{
			name: "positive values",
			pos:  basetypes.Position{X: 10, Z: 20, Y: 30},
			want: nil, // Verified via round-trip
		},
		{
			name: "negative values",
			pos:  basetypes.Position{X: -10, Z: -20, Y: -30},
			want: nil, // Verified via round-trip
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			var buf bytes.Buffer
			n, err := tt.pos.WriteTo(&buf)
			if err != nil {
				t.Fatalf("WriteTo failed: %v", err)
			}
			if n != 8 {
				t.Errorf("WriteTo wrote %d bytes, want 8", n)
			}

			if tt.want != nil {
				got := buf.Bytes()
				if !bytes.Equal(got, tt.want) {
					t.Errorf("WriteTo = %02x, want %02x", got, tt.want)
				}
			}

			// Round-trip test
			var pos2 basetypes.Position
			n2, err := pos2.ReadFrom(&buf)
			if err != nil {
				t.Fatalf("ReadFrom failed: %v", err)
			}
			if n2 != 8 {
				t.Errorf("ReadFrom read %d bytes, want 8", n2)
			}
			if pos2.X != tt.pos.X || pos2.Z != tt.pos.Z || pos2.Y != tt.pos.Y {
				t.Errorf("ReadFrom = {X:%d, Z:%d, Y:%d}, want {X:%d, Z:%d, Y:%d}",
					pos2.X, pos2.Z, pos2.Y, tt.pos.X, tt.pos.Z, tt.pos.Y)
			}
		})
	}
}
`
```

```go
// TestByteCountAccuracy verifies byte counts are accurate
func TestByteCountAccuracy(t *testing.T) {
	tests := []struct {
		name        string
		writeFunc   func(*bytes.Buffer) (int64, error)
		expectedMin int64
		expectedMax int64
	}{
		{
			name: "VarInt small",
			writeFunc: func(buf *bytes.Buffer) (int64, error) {
				v := pk.VarInt(10)
				return v.WriteTo(buf)
			},
			expectedMin: 1,
			expectedMax: 1,
		},
		{
			name: "Float",
			writeFunc: func(buf *bytes.Buffer) (int64, error) {
				v := pk.Float(1.5)
				return v.WriteTo(buf)
			},
			expectedMin: 4,
			expectedMax: 4,
		},
		{
			name: "Double",
			writeFunc: func(buf *bytes.Buffer) (int64, error) {
				v := pk.Double(2.5)
				return v.WriteTo(buf)
			},
			expectedMin: 8,
			expectedMax: 8,
		},
		{
			name: "String short",
			writeFunc: func(buf *bytes.Buffer) (int64, error) {
				v := pk.String("test")
				return v.WriteTo(buf)
			},
			expectedMin: 5,
			expectedMax: 6,
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			var buf bytes.Buffer
			bytesWritten, err := tt.writeFunc(&buf)
			require.NoError(t, err)

			assert.GreaterOrEqual(t, bytesWritten, tt.expectedMin)
			assert.LessOrEqual(t, bytesWritten, tt.expectedMax)
			assert.Equal(t, int64(buf.Len()), bytesWritten)
		})
	}
}
```

```go
package generator

import (

// generateTests generates test files for a version
func generateTests(baseDir, version string, files map[string]string) error {
	pkgName := versionToPkg(version)
	
	// Determine which types are available in this version
	testData := TestGenerationData{
		Version:           version,
		PackageName:       pkgName,
		HasStepTick:       hasType(baseDir, "play/clientbound", "StepTick"),
		HasEntityTeleport: hasType(baseDir, "play/clientbound", "EntityTeleport"),
	}
	
	// Generate bitfield test
	if err := generateBitfieldTest(baseDir, testData); err != nil {
		return fmt.Errorf("failed to generate bitfield test: %w", err)
	}
	
	// Generate integration test
	if err := generateIntegrationTest(baseDir, testData); err != nil {
		return fmt.Errorf("failed to generate integration test: %w", err)
	}
	
	return nil
}

func generateBitfieldTest(baseDir string, data TestGenerationData) error {
	testFile, err := os.Create(filepath.Join(baseDir, "bitfield_test.go"))
	if err != nil {
		return err
	}
	defer testFile.Close()
	
	tmpl := template.Must(template.New("bitfield").Parse(bitfieldTestTemplate))
	return tmpl.Execute(testFile, data)
}

func generateIntegrationTest(baseDir string, data TestGenerationData) error {
	testFile, err := os.Create(filepath.Join(baseDir, "integration_test.go"))
	if err != nil {
		return err
	}
	defer testFile.Close()
	
	tmpl := template.Must(template.New("integration").Parse(integrationTestTemplate))
	return tmpl.Execute(testFile, data)
}

// hasType checks if a type exists in the generated code
func hasType(baseDir, subpackage, typeName string) bool {
	// Check if the file exists and contains the type
	// This is a simple implementation - could be more sophisticated
	file := filepath.Join(baseDir, subpackage, "types.go")
	content, err := os.ReadFile(file)
	if err != nil {
		return false
	}
	return strings.Contains(string(content), "type "+typeName+" struct")
}
```
